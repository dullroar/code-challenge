CREATE TABLE "TimeCards" (
    "Id" INTEGER NOT NULL CONSTRAINT "PK_TimeCards" PRIMARY KEY AUTOINCREMENT,
    "Category" TEXT NOT NULL,
    "Hours" REAL NOT NULL,
    "UserId" TEXT NOT NULL,
    "Year" INTEGER NOT NULL,
    "Week" INTEGER NOT NULL,
    "IsDemo" INTEGER NOT NULL
);
CREATE TRIGGER DemoGuard AFTER INSERT ON TimeCards WHEN (SELECT COALESCE(SUM(Hours),0) FROM TimeCards WHERE Id < NEW.Id) > 40 BEGIN DELETE FROM TimeCards WHERE Id = NEW.Id; END;
